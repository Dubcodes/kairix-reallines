cmake_minimum_required(VERSION 3.13)
# Same 63-byte A5 5A protocol as the already-flashed firmware.
# Works with a Pico SDK set by the VS Code extension or via PICO_SDK_PATH.
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/pico_sdk_import.cmake")
  include("${CMAKE_CURRENT_LIST_DIR}/pico_sdk_import.cmake")
elseif(DEFINED ENV{PICO_SDK_PATH})
  include("$ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake")
else()
  message(FATAL_ERROR "Set PICO_SDK_PATH or supply pico_sdk_import.cmake")
endif()
project(kairix_reallines_pico C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(reallines_pico main.c)
pico_generate_pio_header(reallines_pico ${CMAKE_CURRENT_LIST_DIR}/quadrature_encoder.pio)
target_link_libraries(reallines_pico pico_stdlib hardware_pio hardware_uart hardware_watchdog pico_rand)
pico_enable_stdio_usb(reallines_pico 1)
pico_enable_stdio_uart(reallines_pico 0)
pico_set_program_name(reallines_pico "Kairix RealLines Pico")
pico_set_program_version(reallines_pico "1.0.0")
pico_add_extra_outputs(reallines_pico)
