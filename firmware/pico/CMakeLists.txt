cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED ENV{PICO_SDK_PATH})
  message(FATAL_ERROR "Set PICO_SDK_PATH to your installed official Pico SDK")
endif()
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(kairix_reallines_pico C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(reallines_pico main.c)
pico_generate_pio_header(reallines_pico ${CMAKE_CURRENT_LIST_DIR}/quadrature_encoder.pio)
target_link_libraries(reallines_pico pico_stdlib pico_rand hardware_pio hardware_uart hardware_watchdog)
pico_enable_stdio_usb(reallines_pico 1)
pico_enable_stdio_uart(reallines_pico 0)
pico_add_extra_outputs(reallines_pico)
