cmake_minimum_required(VERSION 3.16)
project(kairix_native_renderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(kairix_native_core src/core.cpp)
target_include_directories(kairix_native_core PUBLIC include)
target_compile_options(kairix_native_core PRIVATE -Wall -Wextra -Wpedantic)

option(KAIRIX_BUILD_SDL_RENDERER "Build the SDL2/KMSDRM renderer" ON)
if(KAIRIX_BUILD_SDL_RENDERER)
  find_package(SDL2 REQUIRED)
  add_executable(kairix-native-renderer src/main.cpp)
  target_link_libraries(kairix-native-renderer PRIVATE kairix_native_core SDL2::SDL2)
  target_compile_options(kairix-native-renderer PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS kairix-native-renderer RUNTIME DESTINATION bin)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(kairix-native-core-tests tests/core_tests.cpp)
  target_link_libraries(kairix-native-core-tests PRIVATE kairix_native_core)
  target_compile_definitions(kairix-native-core-tests PRIVATE KAIRIX_PROJECTION_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/../tests/fixtures/native_projection.csv")
  add_test(NAME native-core COMMAND kairix-native-core-tests)
endif()
